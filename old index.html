create table vocab_words (
  id text primary key,
  word text not null,
  lang text default 'KO',
  folder text default '기본 폴더',
  definition text,
  hanja_origin text,
  synonyms jsonb default '[]'::jsonb,
  antonyms jsonb default '[]'::jsonb,
  nuance_guide text,
  examples jsonb default '[]'::jsonb,
  user_memo text default '',
  is_output_done boolean default false,
  created_at bigint
);
alter table vocab_words enable row level security;
create policy "Public Access" on vocab_words for all using (true) with check (true);
